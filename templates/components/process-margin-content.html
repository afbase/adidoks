"] %} {% set_global in_paragraph = true %} {% endif %} {% set_global current_text = [...current_text, trimmed_line] %} {% else %} {# Empty line - close paragraph if we're in one #} {% if in_paragraph %} {% set_global current_text = [...current_text, "

"] %} {% set_global in_paragraph = false %} {% endif %} {% endif %} {% endif %} {% endfor %} {# Don't forget the last section and close any open paragraph #} {% if in_paragraph %} {% set current_text = [...current_text, "

"] %} {% endif %} {% if current_text %} {% set texts = [...texts, current_text | join(sep="\n")] %} {% endif %} {# Render all sections #} {% for i in range(end=titles | length) %}

{% endfor %} {% endcomponent process_margin_content %}
